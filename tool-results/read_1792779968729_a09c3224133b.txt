    45→/**
    46→ * Web3Forms access key (safe to expose client-side — it can only be used to
    47→ * deliver messages to the business inbox, which is the whole point). The
    48→ * free plan requires the submission to come from the visitor's browser,
    49→ * so this fetch happens directly in the client.
    50→ */
    51→const WEB3FORMS_KEY = process.env.NEXT_PUBLIC_WEB3FORMS_ACCESS_KEY ?? "";
    52→
    53→async function submitToWeb3Forms(
    54→  values: ContactFormValues,
    55→  reference: string
    56→): Promise<boolean> {
    57→  if (!WEB3FORMS_KEY) return false;
    58→  try {
    59→    const res = await fetch("https://api.web3forms.com/submit", {
    60→      method: "POST",
    61→      headers: { "Content-Type": "application/json", Accept: "application/json" },
    62→      body: JSON.stringify({
    63→        access_key: WEB3FORMS_KEY,
    64→        subject: `New website enquiry — ${values.businessName} (#${reference.slice(-8).toUpperCase()})`,
    65→        from_name: "FABOM GLOBAL Website",
    66→        full_name: values.fullName,
    67→        email: values.email,
    68→        business_name: values.businessName,
    69→        current_website: values.currentWebsite || "—",
    70→        business_type: values.businessType,
    71→        service_required: values.serviceRequired,
    72→        estimated_budget: values.estimatedBudget || "—",
    73→        preferred_timeline: values.preferredTimeline || "—",
    74→        message: values.projectDescription,
    75→        reference,
    76→        botcheck: "",
    77→      }),
    78→    });
    79→    const data = (await res.json().catch(() => null)) as { success?: boolean } | null;
    80→    return res.ok && data?.success === true;
    81→  } catch {
    82→    return false;
    83→  }
    84→}
